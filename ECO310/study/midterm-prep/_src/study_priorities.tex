\section{Key Formulas \& Concepts}

\subsection*{Perfect Complements: $u(x_1, x_2) = \min\{\alpha x_1, \beta x_2\}$}

\textbf{Optimal spending:} $\alpha x_1 = \beta x_2$

\textbf{Marshallian:} $x_1 = \frac{\beta m}{\alpha p_1 + \beta p_2}$, $x_2 = \frac{\alpha m}{\alpha p_1 + \beta p_2}$

\textbf{Hicksian:} $x_1^c = \frac{\beta u^*}{\alpha}$, $x_2^c = \frac{\alpha u^*}{\beta}$

\textbf{Expenditure:} $e(p_1, p_2, u^*) = \left(\frac{\beta p_1}{\alpha} + p_2\right) \cdot \frac{\alpha u^*}{\beta}$

\textbf{Key insight:} Substitution effect = 0 (no reoptimization possible)

\subsection*{Cobb-Douglas: $u(x_1, x_2) = x_1^\alpha x_2^\beta$}

\textbf{Magic rule:} Spend $\frac{\alpha}{\alpha + \beta}$ of income on good 1

\subsection*{Perfect Substitutes: $u(x_1, x_2) = \alpha x_1 + \beta x_2$}

\textbf{Bang-per-buck:} Buy only good 1 if $\frac{\alpha}{p_1} > \frac{\beta}{p_2}$

\subsection*{Risk Aversion}

\textbf{Definition:} $u(E[W]) > EU$ (utility of expected wealth beats expected utility)

\textbf{Implies:} $u''(w) \leq 0$ (concave utility)

\textbf{Certainty equivalent:} $u(w_0 + CE) = EU$

\textbf{Risk premium:} $\pi = E[W] - CE$

\subsection*{Portfolio Optimization}

\textbf{EU maximization:} Choose $x$ to maximize $EU(x)$

\textbf{Interior solution:} $EU'(x) = 0$

\textbf{Corner solution:} If $EU'(0) < 0$, invest nothing; if $EU'(x_{\max}) > 0$, invest maximum
