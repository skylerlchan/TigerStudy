\section*{1.3 Probability Axioms}
%----------------------------------------------
\textbf{Axioms:} For probability measure $P$ on $(\Omega, \mathcal{F})$:
\begin{enumerate}[nosep, leftmargin=1.2em]
  \item $P(\Omega) = 1$
  \item $P(A) \geq 0$ for all events $A$
  \item (Countable additivity) If $A_1, A_2, \ldots$ are disjoint:
  \[
    P\!\left(\bigcup_{i=1}^{\infty} A_i\right) = \sum_{i=1}^{\infty} P(A_i)
  \]
\end{enumerate}

\textbf{Consequences:}
\[
  P(A^c) = 1 - P(A)
\]
\[
  P(A \cup B) = P(A) + P(B) - P(A \cap B)
\]
\[
  A \subseteq B \implies P(A) \leq P(B)
\]

\textbf{Inclusion--Exclusion:}
\[
  P\!\left(\bigcup_{i=1}^{n} A_i\right) = \sum_i P(A_i) - \sum_{i<j} P(A_i \cap A_j) + \cdots
\]

\textbf{Union Bound:}
\[
  P\!\left(\bigcup_{i=1}^{n} A_i\right) \leq \sum_{i=1}^{n} P(A_i)
\]

\textbf{Uniform (equally likely):} $|\Omega| < \infty$:
\[
  P(A) = \frac{|A|}{|\Omega|}
\]

\section*{1.4 Counting}
%----------------------------------------------
\textbf{Permutations} of $k$ from $n$: $\displaystyle \frac{n!}{(n-k)!}$

\textbf{Combinations:}
$\displaystyle \binom{n}{k} = \frac{n!}{k!(n-k)!}$

\textbf{Multinomial:} $\displaystyle \binom{n}{k_1, k_2, \ldots, k_r} = \frac{n!}{k_1! k_2! \cdots k_r!}$

\section*{1.5 Conditional Probability}
%----------------------------------------------

\[
  P(A \mid B) = \frac{P(A \cap B)}{P(B)}, \quad P(B) > 0
\]

\textbf{Multiplication Rule:}
\[
  P(A \cap B) = P(A \mid B)\,P(B)
\]

\textbf{Chain rule:}
\[
  P(A_1 \cap \cdots \cap A_n) = P(A_1)\,P(A_2 \mid A_1)\,P(A_3 \mid A_1 \cap A_2) \cdots
\]

\textbf{Law of Total Probability:}
If $B_1, \ldots, B_n$ partition $\Omega$:
\[
  P(A) = \sum_{i=1}^{n} P(A \mid B_i)\,P(B_i)
\]

\textbf{Bayes' Rule:}
\[
  P(B_i \mid A) = \frac{P(A \mid B_i)\,P(B_i)}{\displaystyle\sum_{j} P(A \mid B_j)\,P(B_j)}
\]

%----------------------------------------------
\section*{1.6 Independence}
%----------------------------------------------

$A, B$ independent $\iff$ $P(A \cap B) = P(A)\,P(B)$

Equivalently: $P(A \mid B) = P(A)$

\textbf{Mutual independence:} For \emph{every} subcollection $S \subseteq \{1,\ldots,n\}$:
\[
  P\!\left(\bigcap_{i \in S} A_i\right) = \prod_{i \in S} P(A_i)
\]

Note: pairwise independence $\neq$ mutual independence.

\section*{1.7--1.8 Random Variables \& Expectation}
%----------------------------------------------

\textbf{Indicator r.v.:} $\mathbf{1}_A(\omega) = \begin{cases} 1 & \omega \in A \\ 0 & \omega \notin A \end{cases}$

$P\{\mathbf{1}_A = 1\} = P\{A\}$, \quad $E[\mathbf{1}_A] = P\{A\}$

\textbf{Expectation (discrete):}
\[
  E[X] = \sum_{x} x\,P\{X = x\}
\]

\textbf{Linearity of expectation} (always holds):
\[
  E[aX + bY] = a\,E[X] + b\,E[Y]
\]

\textbf{Function of r.v.:}
\[
  E[g(X)] = \sum_{x} g(x)\,P\{X = x\}
\]

%----------------------------------------------
\section*{1.9 Independence \& Conditioning of R.V.s}
%----------------------------------------------

$X, Y$ independent $\iff$ for all $x, y$:
\[
  P\{X = x,\, Y = y\} = P\{X = x\}\,P\{Y = y\}
\]

If $X \perp Y$: $E[XY] = E[X]\,E[Y]$

\textbf{Conditional expectation:}
\[
  E[X \mid B] = \sum_{x} x\,P\{X = x \mid B\}
\]

\textbf{Tower property:}
\[
  E[X] = \sum_{i} E[X \mid B_i]\,P(B_i)
\]

\section*{2.1 Binomial Distribution}
%----------------------------------------------

$n$ independent Bernoulli$(p)$ trials, $q = 1 - p$.

$S_n = $ number of successes $\sim \text{Binom}(n, p)$:
\[
  P\{S_n = k\} = \binom{n}{k} p^k q^{n-k}, \quad k = 0, 1, \ldots, n
\]
\[
  E[S_n] = np
\]

\textbf{Binomial theorem:}
$\displaystyle (p + q)^n = \sum_{k=0}^{n} \binom{n}{k} p^k q^{n-k} = 1$

%----------------------------------------------
\section*{2.2 Geometric \& Negative Binomial}
%----------------------------------------------

\textbf{Geometric:} $T_1 = $ trial of first success.

$T_1 \sim \text{Geom}(p)$:
\[
  P\{T_1 = n\} = q^{n-1}\,p, \quad n = 1, 2, 3, \ldots
\]
\[
  E[T_1] = \frac{1}{p}
\]

Verification: $\displaystyle\sum_{n=1}^{\infty} q^{n-1} p = \frac{p}{1-q} = 1$

\textbf{Murphy's law:} $P\{T_1 < \infty\} = 1$ (success is guaranteed if $p > 0$).

\textbf{Memoryless property:}
\[
  P\{T_1 > m + n \mid T_1 > m\} = P\{T_1 > n\}
\]

\textbf{Negative Binomial:} $T_k = $ trial of $k$-th success.
\[
  P\{T_k = n\} = \binom{n-1}{k-1} p^k q^{n-k}, \quad n \geq k
\]
\[
  E[T_k] = \frac{k}{p}
\]

Inter-arrival times $T_i - T_{i-1}$ are i.i.d.\ $\text{Geom}(p)$.

\textbf{Geometric series:}
$\displaystyle \sum_{k=0}^{\infty} r^k = \frac{1}{1-r}, \quad |r| < 1$

$\displaystyle \sum_{k=0}^{\infty} k\,r^k = \frac{r}{(1-r)^2}$

\section*{2.3 Law of Large Numbers}
%----------------------------------------------

\textbf{Variance:}
\[
  \text{Var}(X) = E\!\left[(X - E[X])^2\right] = E[X^2] - (E[X])^2
\]

\textbf{Properties:}
\begin{itemize}[nosep, leftmargin=1.2em]
  \item $\text{Var}(X) \geq 0$; equals $0$ iff $X$ is constant a.s.
  \item $\text{Var}(aX) = a^2 \text{Var}(X)$
  \item $\text{Var}(X + c) = \text{Var}(X)$
  \item If $X \perp Y$: $\text{Var}(X + Y) = \text{Var}(X) + \text{Var}(Y)$
\end{itemize}

\textbf{Bernoulli variance:} $X \sim \text{Bern}(p)$: $\text{Var}(X) = p(1-p)$

\textbf{Binomial variance:} $S_n \sim \text{Binom}(n,p)$: $\text{Var}(S_n) = np(1-p)$

\textbf{Chebyshev's inequality:}
\[
  P\{|X - E[X]| \geq a\} \leq \frac{\text{Var}(X)}{a^2}
\]

\textbf{Law of Large Numbers:}
\[
  E\!\left[\left(\frac{S_n}{n} - p\right)^{\!2}\right] = \frac{p(1-p)}{n} \to 0
\]

For i.i.d.\ $X_1, X_2, \ldots$ with $E[X_i] = \mu$:
\[
  \frac{X_1 + X_2 + \cdots + X_n}{n} \xrightarrow{\text{}} \mu \quad \text{as } n \to \infty
\]
